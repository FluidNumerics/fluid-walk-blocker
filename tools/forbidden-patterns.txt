# STRUCTURAL shapes of a site fact. No customer term belongs in this file --
# those live outside the tree (tools/README-ip-gate.md). Python `re` syntax,
# one `name: regex` per line, matched per line. `allow CATEGORY GLOB` exempts
# paths from one category; `allow * GLOB` from all of them.
ipv4: \b(?:\d{1,3}\.){3}\d{1,3}\b
uid-literal: \buid[=:]\s?\d{5,}\b|\buser-\d{5,}\.slice\b
site-hostname: \b[a-z]{3}-[a-z]{2,}-login\b
slurm-jobid: (?i)(?:\bslurm-|\bsacct -j |\bscontrol show job |\bjob )\d{6,8}\b
big-count: \b\d{1,3}(?:,\d{3}){2,}\b
storage-capacity: \b\d+(?:\.\d+)?\s?(?:PiB|PB|EiB|EB)\b
three-decimal-percent: \b\d+\.\d{3}\s?%
snowflake-id: \b\d{9,}\b
measured-date: (?i)\bmeasured\b[^.\n]{0,60}\b20\d\d-\d\d-\d\d\b
slack-channel: (?<![\w&])#[a-z][a-z0-9]*-[a-z0-9-]+\b
github-handle: (?<![\w.@/])@[a-z\d][a-z\d-]{1,38}\b(?![.(\[])
# Decorators and shebangs are not handles or channels.
allow github-handle *.py
allow github-handle *.sh
allow github-handle tools/hooks/*
allow github-handle .github/CODEOWNERS
allow slack-channel *.sh
allow slack-channel tools/hooks/*
# This file and the format exemplar describe the shapes they forbid.
allow * tools/forbidden-patterns.txt
allow * tools/forbidden-terms.example.txt
# The scanner's self-test holds one exemplar per structural shape, on
# purpose; customer terms still apply to it and must never match.
allow * tests/test_ip_gate.py
# The lock file carries package hashes, which look like ids.
allow snowflake-id uv.lock
allow big-count uv.lock
